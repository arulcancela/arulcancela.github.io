# Escenario: Agentes de IA y validación de operaciones con Admin SDK
from googleapiclient.discovery import build
from google.oauth2 import service_account

SCOPES = ["https://www.googleapis.com/auth/admin.directory.user.readonly"]

def audit_2fa_status(service, customer_id="my_customer"):
    """Identifica cuentas sin doble factor antes de aislar permisos."""
    results = service.users().list(customer=customer_id, maxResults=500).execute()
    users = results.get("users", [])
    unprotected = [
        u["primaryEmail"] for u in users
        if not u.get("isEnrolledIn2Sv", False) and not u.get("suspended", False)
    ]
    return unprotected
